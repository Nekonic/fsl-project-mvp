const OUTCOME_WORDS = { held: "lesson.check.held", leaked: "outcome.leaked" }; const DEBRIEF_TONE = { held: "text-success", leaked: "text-danger" }; const toned = (text, tone) => `${esc(text)}`; function debriefSection(name, inner, heading = `lesson.debrief.${name}`) { return `

${esc(t(heading))}

${inner}
`; } function debriefOutcome(outcome) { return outcome ? toned(t(OUTCOME_WORDS[outcome] || "outcome.inconclusive"), DEBRIEF_TONE[outcome] || "text-warning") : `—`; } function debriefChange(change) { const ids = change.ids || []; const idsLines = ids.length ? `

${esc(t("lesson.debrief.change.ids"))}

${esc(ids.join("\n"))}
` : ""; const after = change.after; if (after === null) return `

${esc(t("lesson.debrief.change.unread"))}

${idsLines}`; const lines = []; if (after && after.paranoia !== change.before.paranoia) { lines.push(`

${esc(t("lesson.debrief.change.level", change.before.paranoia ?? "—", after.paranoia ?? "—"))}

`); } if (after && change.diff.length) { lines.push(`
${esc(change.diff.join("\n"))}
`); } if (!lines.length && !ids.length) return `

${esc(t("lesson.debrief.change.none"))}

`; const waf = lines.length ? `

${esc(t("lesson.debrief.change.waf"))}

${lines.join("")}` : ""; return waf + idsLines; } function debriefCustomers(customers) { const actions = Object.entries(customers.blocked_actions || {}).map(([action, count]) => `${localized(`traffic.action.${action}`, action)} ${count}`).join(", "); return [ t("scenario.debrief.customers.line", customers.sent, customers.served, customers.blocked), customers.unanswered || customers.other ? t("scenario.debrief.customers.rest", customers.unanswered || 0, customers.other || 0) : "", ...customers.in_window.map((count, n) => t("scenario.debrief.customers.window", n + 1, count)), actions ? t("scenario.debrief.customers.actions", actions) : "", ].filter(Boolean).map((line) => `

${esc(line)}

`).join(""); } function debriefTimeline(timeline, opened) { return `
    ${timeline.map((entry) => { const after = clock(Date.parse(opened) + entry.at * 1000); const state = entry.state === "fired" ? (entry.outcome ? debriefOutcome(entry.outcome) : "") : toned(t(`scenario.debrief.state.${entry.state}`), entry.state === "failed" ? "text-warning" : "text-muted"); return `
  1. ${esc(after)} ${esc(localized(`case.${entry.case}`, entry.case))} ${entry.origin ? `${esc(t("scenario.debrief.from", localized(`scenario.origin.${entry.origin}`, entry.origin)))}` : ""} ${entry.in_window ? toned(t("scenario.debrief.in_window"), "text-primary") : ""} ${state} ${entry.state === "fired" ? toned(t(entry.detected ? "scenario.debrief.alerted" : "scenario.debrief.no_alert"), entry.detected ? "text-success" : "text-danger") : ""} ${entry.leaked.length ? `${esc(t("scenario.debrief.leaked", entry.leaked.join(", ")))}` : ""}
    2. `; }).join("")}
`; } function renderDebrief(container, told) { const overview = told.overview; const facts = [ overview.verified ? t("scenario.debrief.overview", overview.verified) : "", overview.steps.total ? t("lesson.steps.passed", overview.steps.passed, overview.steps.total) : "", t("scenario.debrief.help", overview.help.hints), overview.fix ? t(`lesson.fix.${overview.fix}`) : "", ].filter(Boolean); const evidence = `
    ${told.evidence.map((row) => `
  • ${esc([row.signature, `${row.method} ${row.path}`.trim(), t(`lesson.debrief.phase.${row.phase}`), localized(`case.${row.case}`, row.case)].filter(Boolean).join(" · "))}
    • `).join("")}
${ told.evidence_total > told.evidence.length ? `

${ esc(t("scenario.debrief.evidence_more", told.evidence_total - told.evidence.length))}

` : ""}`; const alertWord = (alerted) => (alerted == null ? "" : toned(t(alerted ? "scenario.debrief.alerted" : "scenario.debrief.no_alert"), alerted ? "text-success" : "text-danger")); const failedIn = (name, inWindow) => told.timeline.some((entry) => entry.case === name && Boolean(entry.in_window) === inWindow && entry.state === "failed"); const unfired = (name, inWindow) => (failedIn(name, inWindow) ? toned(t("scenario.debrief.state.failed"), "text-warning") : toned(t("scenario.debrief.not_run"), "text-faint")); const judgedCell = (row, inWindow) => { const outcome = inWindow ? row.after : row.before; const alerted = inWindow ? row.alerted_after : row.alerted_before; return [outcome ? debriefOutcome(outcome) : "", alertWord(alerted)].filter(Boolean).join(" · ") || unfired(row.name, inWindow); }; const judgedCases = told.cases.filter((row) => row.before || row.after || row.alerted_before != null || row.alerted_after != null); const cases = `${judgedCases.map((row) => ``).join("")}
${esc(t("score.col.case"))}${esc(t("scenario.debrief.before"))} ${esc(t("scenario.debrief.after"))}
${esc(localized(`case.${row.name}`, row.name))} ${judgedCell(row, false)}${judgedCell(row, true)}
`; container.innerHTML = `

${esc(t("lesson.debrief.heading"))}

` + [ debriefSection("overview", facts.map((line) => `

${esc(line)}

`).join("")), debriefSection("customers", debriefCustomers(told.customers)), debriefSection("timeline", debriefTimeline(told.timeline, told.at.opened)), ...(told.task_class === "judgement" ? [debriefSection("triage", `

${esc(t("scenario.debrief.triage.line", told.triage.right, told.triage.items, told.triage.first_try))}

`)] : []), ...(told.evidence.length ? [debriefSection("evidence", evidence, "scenario.debrief.evidence")] : []), debriefSection("change", debriefChange(told.change)), ...(judgedCases.length ? [debriefSection("cases", cases)] : []), ...(told.risk.length ? [debriefSection("risk", `
    ${told.risk.map(([key, values]) => `
  • ${esc(t(key, ...values))}
    • `).join("")}
`)] : []), ].join(""); container.classList.remove("hidden"); } async function showScenarioDebrief(container) { const told = await api(`/api/sessions/${sessionId()}/scenario/debrief/`); if (told.ok && told.body.overview) renderDebrief(container, told.body); }
