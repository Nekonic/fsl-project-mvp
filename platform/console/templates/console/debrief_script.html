const OUTCOME_WORDS = { held: "lesson.check.held", served: "lesson.check.served", blocked: "outcome.blocked", leaked: "outcome.leaked", inconclusive: "outcome.inconclusive", detected: "outcome.detected", quiet: "outcome.quiet", missed: "outcome.missed", uncorroborated: "outcome.uncorroborated", alerted: "outcome.alerted", }; const DEBRIEF_TONE = { held: "ok", served: "ok", leaked: "red", blocked: "red", inconclusive: "warn", detected: "ok", quiet: "ok", missed: "red", uncorroborated: "warn", alerted: "red", }; const PASSING = ["held", "served", "detected", "quiet"]; function debriefSection(name, inner, heading = `lesson.debrief.${name}`) { return `

${esc(t(heading))}

${inner}
`; } function debriefOutcome(outcome) { return outcome ? badge(t(OUTCOME_WORDS[outcome] || "outcome.inconclusive"), DEBRIEF_TONE[outcome] || "warn") : `—`; } function debriefChange(change) { const after = change.after; if (!after) return `

${esc(t("lesson.debrief.change.unread"))}

`; const lines = []; if (after.paranoia !== change.before.paranoia) { lines.push(`

${esc(t("lesson.debrief.change.level", change.before.paranoia ?? "—", after.paranoia ?? "—"))}

`); } if (change.diff.length) { lines.push(`
${esc(change.diff.join("\n"))}
`); } return lines.join("") || `

${esc(t("lesson.debrief.change.none"))}

`; } function debriefCustomers(customers) { const actions = Object.entries(customers.blocked_actions || {}).map(([action, count]) => `${localized(`traffic.action.${action}`, action)} ${count}`).join(", "); return [ t("scenario.debrief.customers.line", customers.sent, customers.served, customers.blocked), ...customers.in_window.map((count, n) => t("scenario.debrief.customers.window", n + 1, count)), actions ? t("scenario.debrief.customers.actions", actions) : "", ].filter(Boolean).map((line) => `

${esc(line)}

`).join(""); } function debriefTimeline(timeline) { return `
    ${timeline.map((entry) => { const after = `${Math.floor(entry.at / 60)}:${String(entry.at % 60).padStart(2, "0")}`; const state = entry.state === "fired" ? debriefOutcome(entry.outcome) : badge(t(`scenario.debrief.state.${entry.state}`), entry.state === "failed" ? "warn" : "neutral"); return `
  1. ${esc(after)} ${esc(localized(`case.${entry.case}`, entry.case))} ${entry.origin ? `${esc(t("scenario.debrief.from", entry.origin))}` : ""} ${entry.in_window ? badge(t("scenario.debrief.in_window"), "blue") : ""} ${state} ${entry.state === "fired" ? badge(t(entry.detected ? "scenario.debrief.alerted" : "scenario.debrief.no_alert"), entry.detected ? "ok" : "red") : ""} ${entry.leaked.length ? `${esc(t("scenario.debrief.leaked", entry.leaked.join(", ")))}` : ""}
    2. `; }).join("")}
`; } function renderDebrief(container, told) { const overview = told.overview; const ran = told.scenario ? t("scenario.debrief.overview", overview.verified) : t("lesson.debrief.overview.line", overview.verified); const facts = [ ran + (overview.mode === "self" ? ` · ${t("kind.practice")}` : ""), overview.steps.total ? t("lesson.steps.passed", overview.steps.passed, overview.steps.total) : "", told.scenario ? t("scenario.debrief.help", overview.help.hints) : t("lesson.debrief.help", overview.help.hints, overview.help.skipped, overview.help.done_for_me), overview.fix ? t(`lesson.fix.${overview.fix}`) : "", ].filter(Boolean); const evidence = told.evidence.length ? `
    ${told.evidence.map((row) => `
  • ${esc([row.signature, `${row.method} ${row.path}`.trim(), t(`lesson.debrief.phase.${row.phase}`), localized(`case.${row.case}`, row.case)].filter(Boolean).join(" · "))}
    • `).join("")}
` : `

${esc(t("lesson.debrief.evidence.none"))}

`; const cases = `${told.cases.map((row) => ``).join("")}
${esc(t("score.col.case"))}${esc(t(told.scenario ? "scenario.debrief.before" : "lesson.debrief.before"))} ${esc(t(told.scenario ? "scenario.debrief.after" : "lesson.debrief.after"))}
${esc(localized(`case.${row.name}`, row.name))} ${debriefOutcome(row.before)}${debriefOutcome(row.after)}
`; container.innerHTML = `

${esc(t("lesson.debrief.heading"))}

` + [ debriefSection("overview", facts.map((line) => `

${esc(line)}

`).join("")), ...(told.customers ? [debriefSection("customers", debriefCustomers(told.customers))] : []), ...(told.timeline ? [debriefSection("timeline", debriefTimeline(told.timeline))] : []), ...(told.task_class === "judgement" ? [debriefSection("triage", `

${esc(t("scenario.debrief.triage.line", told.triage.right, told.triage.items, told.triage.first_try))}

`)] : []), debriefSection("evidence", evidence, told.scenario ? "scenario.debrief.evidence" : "lesson.debrief.evidence"), debriefSection("change", debriefChange(told.change)), debriefSection("cases", cases), debriefSection("risk", `
    ${told.risk.map(([key, values]) => `
  • ${esc(t(key, ...values))}
    • `).join("")}
`), ].join(""); container.classList.remove("hidden"); } async function showDebrief(container) { const told = await api(`/api/sessions/${sessionId()}/lesson/debrief/`); if (told.ok && told.body.overview && told.body.overview.passed) renderDebrief(container, told.body); } async function showScenarioDebrief(container) { const told = await api(`/api/sessions/${sessionId()}/scenario/debrief/`); if (told.ok && told.body.overview) renderDebrief(container, told.body); }
