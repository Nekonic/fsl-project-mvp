const site = { origin: "", base: "", host: "", history: [], at: -1, expecting: false, heard: [] }; function readable(path) { try { return decodeURIComponent(path); } catch (error) { return path; } } function siteTyped(text) { let path = String(text).trim().replace(/^[a-z]+:\/\//i, ""); if (site.host && path.toLowerCase().startsWith(site.host)) path = path.slice(site.host.length); return path.startsWith("/") ? path : `/${path}`; } function siteShow() { field("site-address").value = `http://${site.host}${readable(site.history[site.at] || "/")}`; field("site-back").disabled = site.at <= 0; field("site-forward").disabled = site.at >= site.history.length - 1; } function siteLoad() { site.expecting = true; field("site-frame").src = site.origin + site.base + site.history[site.at].slice(1); siteShow(); } function siteGo(path) { site.history = site.history.slice(0, site.at + 1).concat([path]); site.at = site.history.length - 1; siteLoad(); } function siteStep(by) { const next = site.at + by; if (next < 0 || next >= site.history.length) return; site.at = next; siteLoad(); } function siteHeard(event) { const said = event.data && event.data.fslSite; const from = field("site-frame").contentWindow; if (typeof said !== "string" || (from && event.source !== from) || !site.base) return; if (event.origin && event.origin !== site.origin) return; if (!said.startsWith(site.base.slice(0, -1))) return; const path = said.slice(site.base.length - 1) || "/"; if (site.expecting) site.history[site.at] = path; else if (path !== site.history[site.at]) { site.history = site.history.slice(0, site.at + 1).concat([path]); site.at = site.history.length - 1; } site.expecting = false; siteShow(); site.heard.forEach((listener) => listener(path)); } function siteBrowser(served) { const origin = new URL(location.href); origin.port = String(served.port); site.origin = origin.origin; site.base = served.path; site.host = served.path.split("/")[3] || ""; document.querySelector('.pane-tab[data-pane="site"]').classList.remove("hidden"); field("site-bar").addEventListener("submit", (event) => { event.preventDefault(); siteGo(siteTyped(field("site-address").value)); }); field("site-back").addEventListener("click", () => siteStep(-1)); field("site-forward").addEventListener("click", () => siteStep(1)); field("site-reload").addEventListener("click", () => { if (site.history.length) siteLoad(); }); window.addEventListener("message", siteHeard); devTools(); }
